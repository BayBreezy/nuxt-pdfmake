```vue [TableExample.vue]
<template>
  <main class="h-175 w-full">
    <UiIframeLazy v-if="pdfLink" :src="pdfLink" class="h-full w-full" />
  </main>
</template>

<script setup lang="ts">
const pdfLink = ref<string | null>(null);

const loadPdf = async () => {
  const pdfMake = usePDFMake();
  if (!pdfMake) return;

  pdfMake.addTableLayouts({
    striped: {
      fillColor: (rowIndex: number) =>
        rowIndex === 0 ? "#1d293d" : rowIndex % 2 === 0 ? "#f8fafc" : null,
      hLineWidth: (rowIndex: number) => (rowIndex === 0 ? 0 : 0.5),
      vLineWidth: () => 0,
      hLineColor: "#e2e8f0",
      paddingBottom: () => 8,
      paddingTop: () => 8,
      paddingLeft: () => 10,
      paddingRight: () => 10,
    },
  });

  pdfLink.value = await pdfMake
    .createPdf({
      info: {
        title: "Table Features Reference",
        author: "Acme Engineering",
        subject: "pdfMake table capabilities overview",
        keywords: "tables, spans, layouts, pdfmake",
        creator: "Nuxt pdfMake",
        producer: "pdfmake",
      },
      pageMargins: [40, 46, 40, 42],
      content: [
        {
          columns: [
            [
              { text: "Table Features Reference", style: "title" },
              { text: "A practical guide to pdfMake table capabilities", style: "subtitle" },
            ],
            {
              text: `Generated ${new Date().toLocaleDateString()}`,
              style: "meta",
              alignment: "right",
            },
          ],
          marginBottom: 22,
        },
        { text: "1. Named Layout via addTableLayouts()", style: "sectionTitle" },
        {
          text: "Register a reusable layout once and reference it by name across the document.",
          style: "bodyText",
          marginBottom: 8,
        },
        {
          layout: "striped",
          table: {
            headerRows: 1,
            widths: ["*", 90, 70, 70],
            body: [
              ["Product", "Category", "Stock", "Price"].map((h) => ({ text: h, style: "th" })),
              ["Wireless Keyboard", "Peripherals", "142", "$89.00"],
              ["USB-C Hub (7-port)", "Peripherals", "58", "$49.00"],
              ["Ergonomic Mouse", "Peripherals", "203", "$65.00"],
              ["Standing Desk Mat", "Furniture", "34", "$110.00"],
              ["Monitor Arm (dual)", "Furniture", "19", "$135.00"],
            ],
          },
          marginBottom: 20,
        },
        { text: "2. Column & Row Spans", style: "sectionTitle" },
        {
          text: "Cells can span multiple columns or rows using colSpan and rowSpan.",
          style: "bodyText",
          marginBottom: 8,
        },
        {
          table: {
            widths: ["*", "*", "*"],
            headerRows: 2,
            body: [
              [
                {
                  text: "Q1 & Q2 Budget",
                  style: "th",
                  colSpan: 2,
                  alignment: "center",
                  fillColor: "#1d293d",
                },
                {},
                { text: "Actuals", style: "th", alignment: "center", fillColor: "#1d293d" },
              ],
              [
                { text: "Q1", style: "subTh", alignment: "center" },
                { text: "Q2", style: "subTh", alignment: "center" },
                { text: "YTD", style: "subTh", alignment: "center" },
              ],
              [
                "$124,000",
                "$138,000",
                {
                  rowSpan: 3,
                  text: "$780,000\n\n(on track)",
                  style: "spanCell",
                  alignment: "center",
                },
              ],
              ["$142,000", "$151,000", ""],
              ["$110,000", "$115,000", ""],
            ],
          },
          layout: {
            hLineColor: () => "#e2e8f0",
            vLineColor: () => "#e2e8f0",
            paddingBottom: () => 8,
            paddingTop: () => 8,
            paddingLeft: () => 10,
            paddingRight: () => 10,
          },
          marginBottom: 20,
        },
        { text: "3. Built-in Layout Presets", style: "sectionTitle" },
        {
          text: "pdfMake ships with noBorders, headerLineOnly, and lightHorizontalLines presets.",
          style: "bodyText",
          marginBottom: 8,
        },
        {
          columns: [
            {
              width: "*",
              stack: [
                { text: "noBorders", style: "presetLabel" },
                {
                  table: {
                    widths: ["*", "*"],
                    body: [
                      [
                        { text: "Name", style: "thSmall" },
                        { text: "Status", style: "thSmall" },
                      ],
                      ["Alice Chen", "Active"],
                      ["Bob Martinez", "On Leave"],
                      ["Carol Singh", "Active"],
                    ],
                  },
                  layout: "noBorders",
                },
              ],
            },
            {
              width: "*",
              stack: [
                { text: "lightHorizontalLines", style: "presetLabel" },
                {
                  table: {
                    widths: ["*", "*"],
                    body: [
                      [
                        { text: "Name", style: "thSmall" },
                        { text: "Status", style: "thSmall" },
                      ],
                      ["Alice Chen", "Active"],
                      ["Bob Martinez", "On Leave"],
                      ["Carol Singh", "Active"],
                    ],
                  },
                  layout: "lightHorizontalLines",
                },
              ],
            },
          ],
          columnGap: 14,
          marginBottom: 20,
        },
        { text: "4. Dashed Borders & Zebra Fill", style: "sectionTitle" },
        {
          text: "Custom layout functions give fine-grained control over line styles and cell backgrounds.",
          style: "bodyText",
          marginBottom: 8,
        },
        {
          table: {
            widths: ["*", "*", "*"],
            body: [
              ["Sprint 23", "Velocity: 42 pts", "Status: Completed"],
              ["Sprint 24", "Velocity: 38 pts", "Status: Completed"],
              ["Sprint 25", "Velocity: 45 pts", "Status: In Progress"],
              ["Sprint 26", "Velocity: — pts", "Status: Planned"],
            ],
          },
          layout: {
            hLineWidth: (i: number, node: any) =>
              i === 0 || i === node.table.body.length ? 1.5 : 0.5,
            vLineWidth: () => 0.5,
            hLineColor: () => "#1d293d",
            vLineColor: () => "#e2e8f0",
            hLineStyle: (i: number, node: any) => {
              if (i === 0 || i === node.table.body.length) return null;
              return { dash: { length: 6, space: 3 } };
            },
            fillColor: (rowIndex: number) => (rowIndex % 2 === 0 ? "#f8fafc" : null),
            paddingBottom: () => 7,
            paddingTop: () => 7,
            paddingLeft: () => 10,
            paddingRight: () => 10,
          },
        },
      ],
      defaultStyle: {
        color: "#1d293d",
        fontSize: 9,
      },
      styles: {
        title: {
          fontSize: 20,
          bold: true,
          color: "#0f172a",
        },
        subtitle: {
          fontSize: 10,
          color: "#64748b",
          marginTop: 3,
        },
        meta: {
          fontSize: 9,
          color: "#64748b",
        },
        sectionTitle: {
          fontSize: 12,
          bold: true,
          color: "#0f172a",
          marginBottom: 4,
        },
        bodyText: {
          color: "#64748b",
          fontSize: 9,
        },
        th: {
          bold: true,
          color: "#ffffff",
          fontSize: 8,
        },
        subTh: {
          bold: true,
          color: "#475569",
          fontSize: 8,
          fillColor: "#f1f5f9",
        },
        spanCell: {
          bold: true,
          color: "#16a34a",
          fontSize: 10,
        },
        thSmall: {
          bold: true,
          color: "#0f172a",
          fontSize: 8,
        },
        presetLabel: {
          fontSize: 8,
          color: "#64748b",
          italics: true,
          marginBottom: 4,
        },
      },
    })
    .getDataUrl();
};

onMounted(() => {
  loadPdf();
});
</script>
```
