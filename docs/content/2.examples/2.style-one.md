```vue [StyleOneExample.vue]
<template>
  <main class="h-175 w-full">
    <UiIframeLazy v-if="pdfLink" :src="pdfLink" class="h-full w-full" />
  </main>
</template>

<script setup lang="ts">
const pdfLink = ref<string | null>(null);

const loadPdf = async () => {
  const pdfMake = usePDFMake();
  if (!pdfMake) return;

  pdfLink.value = await pdfMake
    .createPdf({
      info: {
        title: "Customer Success Pulse Report",
        author: "Acme Operations Group",
        subject: "Monthly customer success performance summary",
        keywords: "customer success, report, styles, pdfmake",
        creator: "Nuxt pdfMake",
        producer: "pdfmake",
      },
      pageMargins: [42, 44, 42, 44],
      content: [
        {
          table: {
            widths: ["*"],
            body: [
              [
                {
                  stack: [
                    { text: "Customer Success Pulse", style: "eyebrow" },
                    { text: "Monthly Performance Report", style: "heroTitle" },
                    {
                      text: "A concise view of retention health, support demand, and near-term priorities.",
                      style: "heroSubtitle",
                    },
                  ],
                  fillColor: "#0f172a",
                  margin: [18, 16, 18, 18],
                },
              ],
            ],
          },
          layout: "noBorders",
          marginBottom: 18,
        },
        {
          columns: [
            { text: "Reporting period\nMarch 2026", style: "statCard" },
            { text: "Accounts reviewed\n128", style: "statCard" },
            { text: "Health score\n91%", style: ["statCard", "positive"] },
          ],
          columnGap: 10,
          marginBottom: 18,
        },
        { text: "Executive Summary", style: "sectionTitle" },
        {
          text: "Retention remains strong across managed accounts, with expansion opportunities concentrated in customers using automation and reporting features. Support volume increased slightly after the billing update, but first-response time stayed within target.",
          style: "paragraph",
        },
        {
          columns: [
            [
              { text: "Highlights", style: "subheader" },
              {
                ul: [
                  "Net revenue retention improved to 116%.",
                  "Enterprise onboarding cycle time dropped by 18%.",
                  "Self-serve customers created 2,400 PDF exports this month.",
                ],
                style: "list",
              },
            ],
            [
              { text: "Focus Areas", style: "subheader" },
              {
                ul: [
                  "Reduce billing-related support tickets.",
                  "Improve adoption of saved report templates.",
                  "Schedule executive reviews for at-risk accounts.",
                ],
                style: "list",
              },
            ],
          ],
          columnGap: 18,
          marginTop: 8,
          marginBottom: 18,
        },
        {
          text: "Style arrays are useful for combining reusable formatting. This note uses both the callout and small styles.",
          style: ["callout", "small"],
        },
        { text: "Next Actions", style: "sectionTitle", marginTop: 18 },
        {
          table: {
            headerRows: 1,
            widths: ["*", 86, 78],
            body: [
              [
                { text: "Action", style: "tableHeader" },
                { text: "Owner", style: "tableHeader" },
                { text: "Due", style: "tableHeader" },
              ],
              ["Send billing update follow-up to affected workspaces", "Support", "Apr 5"],
              ["Prepare expansion proposal for automation cohort", "Success", "Apr 12"],
              ["Review report template usage with product team", "Product", "Apr 18"],
            ],
          },
          layout: {
            fillColor: (rowIndex: number) =>
              rowIndex === 0 ? "#1d293d" : rowIndex % 2 === 0 ? "#f8fafc" : null,
            hLineColor: () => "#e2e8f0",
            vLineColor: () => "#e2e8f0",
            paddingBottom: () => 8,
            paddingTop: () => 8,
            paddingLeft: () => 10,
            paddingRight: () => 10,
          },
        },
      ],
      defaultStyle: {
        color: "#1d293d",
        fontSize: 10,
        lineHeight: 1.25,
      },
      styles: {
        eyebrow: {
          color: "#93c5fd",
          fontSize: 9,
          bold: true,
          characterSpacing: 1,
        },
        heroTitle: {
          color: "#ffffff",
          fontSize: 24,
          bold: true,
          marginTop: 6,
        },
        heroSubtitle: {
          color: "#cbd5e1",
          fontSize: 10,
          marginTop: 6,
        },
        statCard: {
          bold: true,
          color: "#0f172a",
          fillColor: "#f8fafc",
          margin: [12, 10, 12, 10],
        },
        positive: {
          color: "#16a34a",
        },
        sectionTitle: {
          fontSize: 14,
          bold: true,
          color: "#0f172a",
          marginBottom: 6,
        },
        subheader: {
          fontSize: 11,
          bold: true,
          color: "#0f172a",
          marginBottom: 5,
        },
        paragraph: {
          color: "#475569",
          marginBottom: 8,
        },
        list: {
          color: "#475569",
          marginBottom: 4,
        },
        callout: {
          italics: true,
          color: "#0f172a",
          fillColor: "#eff6ff",
          margin: [12, 10, 12, 10],
        },
        small: {
          fontSize: 8,
          color: "#64748b",
        },
        tableHeader: {
          bold: true,
          color: "#ffffff",
          fontSize: 8,
        },
      },
    })
    .getDataUrl();
};

onMounted(() => {
  loadPdf();
});
</script>
```
